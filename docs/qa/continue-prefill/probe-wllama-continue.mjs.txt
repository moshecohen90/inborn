import { createRequire } from "node:module"; import { readdirSync, writeFileSync } from "node:fs"; import os from "node:os"; import path from "node:path";
const req = createRequire("/Users/moshecohen/.npm/_npx/9833c18b2d85bc59/node_modules/");
const { chromium } = req("playwright-core");
const cache = path.join(os.homedir(), "Library/Caches/ms-playwright");
const shell = readdirSync(cache).filter(d => /^chromium_headless_shell-\d+$/.test(d)).sort().reverse().map(d => path.join(cache, d, "chrome-headless-shell-mac-arm64", "chrome-headless-shell"))[0];
const browser = await chromium.launch({ headless: true, executablePath: shell });
console.log("browser pid", browser.process?.()?.pid);
const p = await (await browser.newContext()).newPage();
await p.goto("http://127.0.0.1:8931/probe.html");
const ok = await p.evaluate(() => window.__ready);
console.log("ready", ok, await p.evaluate(() => window.__log));
const Q = "Explain in 8 sentences how a sailing ship tacks against the wind.";
const sys = { role: "system", content: "You are a helpful assistant." };
const base = { temperature: 0.7, top_p: 0.9, repeat_penalty: 1.05 };
const out = {};
const run = (r) => p.evaluate((r) => window.__run(r), r);
out.baseline = await run({ ...base, messages: [sys, { role: "user", content: Q }], max_tokens: 90, chat_template_kwargs: { enable_thinking: false } });
const T = out.baseline.text;
let cutAt = Math.min(T.length - 5, 180); while (cutAt < T.length && /\s/.test(T[cutAt])) cutAt++; cutAt += 2; // two letters into a word
const cut = T.slice(0, cutAt);
out.cut = cut;
out.contA = await run({ ...base, messages: [sys, { role: "user", content: Q }, { role: "assistant", content: cut }], max_tokens: 60, chat_template_kwargs: { enable_thinking: false } });
out.contB = await run({ ...base, messages: [sys, { role: "user", content: Q }, { role: "assistant", content: cut }], max_tokens: 60, continue_final_message: true, add_generation_prompt: false, chat_template_kwargs: { enable_thinking: false } });
out.thinkBase = await run({ ...base, messages: [sys, { role: "user", content: Q }], max_tokens: 400, chat_template_kwargs: { enable_thinking: true } });
const TT = out.thinkBase.text; const cut2 = TT.slice(0, Math.min(TT.length, 60));
out.cut2 = cut2;
if (cut2) out.contThink = await run({ ...base, messages: [sys, { role: "user", content: Q }, { role: "assistant", content: cut2, reasoning_content: out.thinkBase.reasoning }], max_tokens: 60, chat_template_kwargs: { enable_thinking: true } });
writeFileSync("probe-out.json", JSON.stringify(out, null, 1));
console.log(JSON.stringify(out, null, 1).slice(0, 6000));
await browser.close();
