// F443 live proof: node live.mjs <base> <outDir> <runs: e.g. 1,2,3,4,5,6,photo>
import { createRequire } from "node:module"; import { readdirSync, mkdirSync, writeFileSync, appendFileSync } from "node:fs"; import os from "node:os"; import path from "node:path";
const req = createRequire("/Users/moshecohen/.npm/_npx/9833c18b2d85bc59/node_modules/");
const { chromium } = req("playwright-core");
const [base, outDir, runsArg = "1,2,3,4,5,6", profile] = process.argv.slice(2);
mkdirSync(outDir, { recursive: true });
const cache = path.join(os.homedir(), "Library/Caches/ms-playwright");
const shell = readdirSync(cache).filter(d => /^chromium_headless_shell-\d+$/.test(d)).sort().reverse().map(d => path.join(cache, d, "chrome-headless-shell-mac-arm64", "chrome-headless-shell"))[0];
const ctx = await chromium.launchPersistentContext(profile, { headless: true, executablePath: shell, viewport: { width: 1280, height: 900 } });
const p = ctx.pages()[0] ?? await ctx.newPage();
const logs = [];
p.on("console", m => { const t = m.text(); logs.push(t); if (/\[wllama\] continue|\[chat\] loop|\[wllama\] loaded|error/i.test(t)) console.log("console:", t.slice(0, 220)); });
const note = (s) => { console.log(s); appendFileSync(path.join(outDir, "notes.txt"), s + "\n"); };
const ids = async () => [...new Set(await p.evaluate(() => [...document.querySelectorAll("[data-testid]")].filter(e => { const r = e.getBoundingClientRect(); const c = getComputedStyle(e); return r.width > 0 && r.height > 0 && c.visibility === "visible" && c.opacity !== "0"; }).map(e => e.getAttribute("data-testid"))))];
const clickId = async (id, t = 8000) => { const ok = await p.evaluate((id) => { const els = [...document.querySelectorAll(`[data-testid="${id}"]`)]; const el = els.find(e => e.getBoundingClientRect().width > 0) || els[0]; if (!el) return false; el.focus(); return document.activeElement === el; }, id); if (ok) { await p.keyboard.press("Enter"); return; } try { await p.locator(`[data-testid="${id}"]`).last().click({ timeout: t, force: true }); } catch (e) { note(`click ${id}: ${String(e).split("\n")[0].slice(0, 100)}`); } };
const mouseId = async (id) => { const r = await p.evaluate((id) => { const els = [...document.querySelectorAll(`[data-testid="${id}"]`)]; const el = els.find(e => e.getBoundingClientRect().width > 0); if (!el) return null; el.scrollIntoView({ block: "center" }); const b = el.getBoundingClientRect(); return [b.x + b.width / 2, b.y + b.height / 2]; }, id); if (!r) return false; await p.mouse.click(r[0], r[1]); return true; };
const lastText = () => p.evaluate(() => { const els = [...document.querySelectorAll('[data-testid="assistant-text"]')]; return els.length ? els[els.length - 1].innerText : ""; });
const shot = async (name) => { const f = path.join(outDir, name + ".png"); await p.screenshot({ path: f, fullPage: false }); return f; };

const ready = async () => {
  await p.goto(base + "/", { waitUntil: "networkidle" }); await p.waitForTimeout(3000);
  const t0 = Date.now(); let dl = false;
  while (Date.now() - t0 < 900000) {
    const cur = await ids();
    if (cur.includes("composer-input")) return true;
    if (cur.includes("onboarding-continue")) { await clickId("onboarding-continue"); await p.waitForTimeout(2500); continue; }
    if (cur.includes("lock-start")) { await clickId("lock-start"); await p.waitForTimeout(3000); continue; }
    if (cur.includes("sealed-start")) { await p.waitForTimeout(4000); await clickId("sealed-start", 30000); await p.waitForTimeout(3000); continue; }
    if (cur.includes("download-model") && !dl) { await clickId("download-model"); dl = true; await p.waitForTimeout(3000); continue; }
    if (cur.includes("use-model")) { await clickId("use-model"); await p.waitForTimeout(3000); continue; }
    await p.waitForTimeout(2000);
  }
  return false;
};
const chip = async () => (await p.locator('[data-testid="model-chip"]').first().innerText().catch(() => "?")).replace(/\s+/g, " ");
const ensureFast = async () => {
  if (/fast/i.test(await chip())) return;
  await mouseId("model-chip"); await p.waitForTimeout(1500);
  if (!(await mouseId("model-sheet-choose-fast"))) { note("no model-sheet-choose-fast; ids " + (await ids()).filter(i => /model-sheet/.test(i)).join(" ")); return; }
  await p.waitForTimeout(3000);
  const t0 = Date.now();
  while (Date.now() - t0 < 600000) { const cur = await ids(); if (cur.includes("model-sheet-confirm-fast")) await mouseId("model-sheet-confirm-fast"); if (cur.includes("download-model")) await clickId("download-model"); if (cur.includes("sealed-start")) { await p.waitForTimeout(3000); await clickId("sealed-start", 30000); } if (cur.includes("lock-start")) await clickId("lock-start"); if (/fast/i.test(await chip()) && cur.includes("composer-input")) break; await p.waitForTimeout(3000); }
  if ((await ids()).includes("model-sheet-close")) await mouseId("model-sheet-close");
};
const newChat = async () => {
  await p.keyboard.press("Meta+n"); await p.waitForTimeout(2000);
  if ((await p.locator('[data-testid="assistant-message"]').count()) === 0) return "meta+n";
  await ready(); await p.waitForTimeout(1000);
  return (await p.locator('[data-testid="assistant-message"]').count()) === 0 ? "reload" : "NOT NEW";
};
const words8 = (s) => s.trim().split(/\s+/).slice(0, 8).join(" ");
const joins = [];
const runOne = async (tag, q, stopAt = 200) => {
  const how = await newChat();
  await p.locator('[data-testid="composer-input"]').fill(q); await p.waitForTimeout(300);
  if (!(await mouseId("send"))) await p.keyboard.press("Enter");
  await p.waitForTimeout(1500);
  if ((await p.locator('[data-testid="assistant-message"]').count()) === 0 && (await p.locator('[data-testid="composer-input"]').inputValue().catch(() => ""))) { note(tag + ": send did not start a turn; retry Enter"); await p.locator('[data-testid="composer-input"]').focus(); await p.keyboard.press("Enter"); }
  const t0 = Date.now(); let len = 0;
  while (Date.now() - t0 < 300000) { len = (await lastText()).length; if (len >= stopAt) break; await p.waitForTimeout(80); }
  await clickId("stop");
  const tStop = Date.now();
  while (Date.now() - tStop < 20000 && !(await ids()).includes("continue")) await p.waitForTimeout(300);
  await p.waitForTimeout(800);
  const stopped = await lastText();
  const logFrom = logs.length;
  await shot(`${tag}-a-stopped`);
  await clickId("continue");
  const t2 = Date.now(); let prev = ""; let same = 0;
  while (Date.now() - t2 < 420000) { await p.waitForTimeout(1500); const cur = await ids(); const tx = await lastText(); if (!cur.includes("stop") && tx.length > stopped.length && tx === prev) { if (++same >= 2) break; } else same = 0; prev = tx; }
  const final = await lastText();
  const f = await shot(`${tag}-b-final`);
  const kept = final.startsWith(stopped.replace(/\s+$/u, ""));
  const cutAt = stopped.replace(/\s+$/u, "").length;
  const after = final.slice(cutAt);
  const lines = logs.slice(logFrom).filter(l => /\[wllama\] continue|\[chat\] loop/.test(l));
  const j = { tag, q, newChat: how, stopped, final, keptOnScreen: kept, seam: `…${stopped.slice(-40)}⟦${after.slice(0, 60)}⟧`, first8: words8(after), console: lines, chip: await chip() };
  joins.push(j);
  note(`${tag} (${how}) kept=${kept} seam: ${JSON.stringify(j.seam)} | ${lines.join(" | ")}`);
  return j;
};

const Q = "Explain in 8 sentences how a sailing ship tacks against the wind.";
const plan = {
  1: ["r1-tack", Q], 2: ["r2-tack", Q], 3: ["r3-tack", Q],
  4: ["r4-list", "Give me a numbered list of 10 tips for staying focused while studying, one sentence each."],
  5: ["r5-story", "Write a short story about a lighthouse keeper who finds a message in a bottle."],
  6: ["r6-howto", "How do I change a flat tyre on a car? Explain it step by step."],
};
note(`start ${new Date().toISOString()} ready=${await ready()} chip=${await chip()}`);
await ensureFast(); note("chip after ensureFast: " + await chip());
for (const k of runsArg.split(",")) {
  if (plan[k]) await runOne(...plan[k]);
  if (k === "photo") {
    await newChat();
    const png = path.join(outDir, "cat.png");
    const buf = await p.evaluate(async () => { const c = document.createElement("canvas"); c.width = 320; c.height = 320; const x = c.getContext("2d"); x.fillStyle = "#fff"; x.fillRect(0,0,320,320); x.fillStyle = "#d22"; x.beginPath(); x.arc(160,120,70,0,Math.PI*2); x.fill(); x.fillStyle = "#000"; x.font = "bold 64px sans-serif"; x.textAlign = "center"; x.fillText("CAT",160,290); return c.toDataURL("image/png"); });
    writeFileSync(png, Buffer.from(buf.split(",")[1], "base64"));
    await clickId("attach"); await p.waitForTimeout(1500);
    const [chooser] = await Promise.all([p.waitForEvent("filechooser", { timeout: 8000 }).catch(() => null), clickId("attach-photo")]);
    if (chooser) await chooser.setFiles(png); else await p.locator('input[type=file]').first().setInputFiles(png).catch(e => note("setInputFiles: " + String(e).slice(0, 100)));
    await p.waitForTimeout(2500);
    note("photo ids after pick: " + (await ids()).filter(i => /attach|photo|vision|chip/.test(i)).join(" "));
    await p.locator('[data-testid="composer-input"]').fill("Describe this picture in detail in six sentences: the shape, its colour, the word under it, and what the picture could be used for.");
    await p.waitForTimeout(300); if (!(await mouseId("send"))) await p.keyboard.press("Enter"); await p.waitForTimeout(4000);
    const hold = (await ids()).find(i => /^vision-hold-(download|primary)$/.test(i)); note("photo hold: " + hold); if (hold) await mouseId(hold);
    const t0 = Date.now(); let len = 0;
    while (Date.now() - t0 < 900000) { len = (await lastText()).length; if (len >= 200) break; const cur = await ids(); if (!cur.includes("stop") && !cur.includes("vision-hold") && len > 0 && Date.now() - t0 > 30000) break; await p.waitForTimeout(100); }
    note(`photo answer reached ${len} chars after ${Math.round((Date.now() - t0) / 1000)} s`);
    if ((await ids()).includes("stop")) await clickId("stop");
    const tStop = Date.now(); while (Date.now() - tStop < 20000 && !(await ids()).includes("continue")) await p.waitForTimeout(300);
    await p.waitForTimeout(800);
    const stopped = await lastText(); await shot("r7-photo-a-stopped");
    const logFrom = logs.length;
    if ((await ids()).includes("continue")) {
      await clickId("continue");
      const t2 = Date.now(); let prev = ""; let same = 0;
      while (Date.now() - t2 < 600000) { await p.waitForTimeout(1500); const cur = await ids(); const tx = await lastText(); if (!cur.includes("stop") && tx.length > stopped.length && tx === prev) { if (++same >= 2) break; } else same = 0; prev = tx; }
    } else note("photo: no continue offered; ids " + (await ids()).join(" "));
    const final = await lastText(); await shot("r7-photo-b-final");
    const cutAt = stopped.replace(/\s+$/u, "").length; const after = final.slice(cutAt);
    const lines = logs.slice(logFrom).filter(l => /\[wllama\] continue|\[chat\] loop|\[wllama\] photo/.test(l));
    joins.push({ tag: "r7-photo", q: "photo (CAT image)", stopped, final, keptOnScreen: final.startsWith(stopped.replace(/\s+$/u, "")), seam: `…${stopped.slice(-40)}⟦${after.slice(0, 60)}⟧`, first8: words8(after), console: lines, chip: await chip() });
    note(`r7-photo kept=${joins.at(-1).keptOnScreen} seam: ${JSON.stringify(joins.at(-1).seam)} | ${lines.join(" | ")}`);
  }
}
writeFileSync(path.join(outDir, `joins-${runsArg.replace(/,/g, "")}.json`), JSON.stringify(joins, null, 1));
writeFileSync(path.join(outDir, "console.txt"), logs.filter(l => /wllama|chat\]/.test(l)).join("\n"));
await ctx.close();
note("done " + new Date().toISOString());
