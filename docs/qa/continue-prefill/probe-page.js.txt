import { Wllama, LoggerWithoutDebug } from "/w/index.js";
const log = (...a) => { window.__log.push(a.map(x => typeof x === "string" ? x : JSON.stringify(x)).join(" ")); };
window.__log = [];
window.__ready = (async () => {
  const w = new Wllama({ default: "/w/wasm/wllama.wasm" }, { logger: LoggerWithoutDebug, allowOffline: true });
  w.setCompat(null);
  const t0 = performance.now();
  await w.loadModelFromUrl("/models/fast.gguf", { n_ctx: 4096, n_threads: Math.max(1, Math.floor(navigator.hardwareConcurrency / 2)), jinja: true, n_gpu_layers: 0 });
  log("loaded ms", Math.round(performance.now() - t0), "threads", w.getNumThreads());
  window.__w = w;
  return true;
})().catch(e => { log("load error", String(e)); return false; });
window.__run = async (req) => {
  const w = window.__w; const chunks = []; let text = "", reasoning = ""; let usage = null, timings = null;
  try {
    for await (const c of await w.createChatCompletion({ ...req, stream: true })) {
      const d = c.choices[0]?.delta ?? {};
      if (chunks.length < 6) chunks.push(JSON.stringify(d).slice(0, 300));
      if (d.content) text += d.content; if (d.reasoning_content) reasoning += d.reasoning_content;
      if (c.usage) usage = c.usage; if (c.timings) timings = c.timings;
    }
  } catch (e) { return { error: String(e?.message ?? e) }; }
  return { text, reasoning, first: chunks, usage, timings };
};
