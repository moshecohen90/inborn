import { it } from "vitest";
import initSqlJs, { type Database } from "sql.js";
import * as rag from "../src/index";
import type { DocumentRecord, Embedder, OpenedDocument, SqlDriver, SqlValue } from "../src/index";

function driverOf(db: Database): SqlDriver {
  const bind = (p?: SqlValue[]) => (p ?? []) as (string | number | null)[];
  return {
    exec: async (sql) => void db.exec(sql),
    run: async (sql, params) => void db.run(sql, bind(params)),
    all: async <T,>(sql: string, params?: SqlValue[]) => { const s = db.prepare(sql); s.bind(bind(params)); const r: T[] = []; while (s.step()) r.push(s.getAsObject() as T); s.free(); return r; },
    batch: async (st) => { db.exec("BEGIN"); for (const s of st) db.run(s.sql, bind(s.params)); db.exec("COMMIT"); },
  };
}
const WORDS = "quay berth tide crane pilot mooring dredging channel ferry cargo gantry wharf breakwater buoy tugboat lighthouse the of and to in".split(" ");
const pageText = (d: number, p: number) => Array.from({ length: 520 }, (_, i) => WORDS[(i * 7 + p * 3 + d) % WORDS.length]).join(" ") + ".";
it("bench", { timeout: 600_000 }, async () => {
  const SQL = await initSqlJs();
  const store = await rag.SqlEmbeddingStore.open(driverOf(new SQL.Database()));
  let engineMs = 0, calls = 0;
  const engine: Embedder = { id: "embed-e5", embed: async (texts) => texts.map((t) => { const s = performance.now(); while (performance.now() - s < 3); engineMs += performance.now() - s; calls++; return rag.hashVector(t, 1024); }) };
  const lanes = (rag as unknown as { EmbedLanes?: new (e: Embedder) => { index: Embedder } }).EmbedLanes;
  const embedder = lanes ? new lanes(engine).index : engine;
  const started = performance.now();
  let chunks = 0;
  for (let d = 0; d < 3; d++) {
    const opened: OpenedDocument = { pages: 40, page: async (i) => ({ page: i + 1, text: pageText(d, i), needsOcr: false }), close: async () => undefined };
    const doc: DocumentRecord = { id: `d${d}`, name: `d${d}.pdf`, kind: "pdf", bytes: 1, pages: 40, addedAt: d, status: "queued", indexedPages: 0, chunkCount: 0, flaggedLines: 0, ocrPages: 0 };
    const r = await rag.indexDocument({ doc, opened, embedder, store, chunk: rag.chunkFor(512) });
    chunks += r.chunkCount;
  }
  const total = performance.now() - started;
  (await import("node:fs")).appendFileSync("bench.txt", `BENCH ${lanes ? "after " : "before"} chunks=${chunks} engineCalls=${calls} total=${total.toFixed(0)}ms engine=${engineMs.toFixed(0)}ms other=${(total - engineMs).toFixed(0)}ms other/chunk=${((total - engineMs) / chunks).toFixed(2)}ms\n`);
});
