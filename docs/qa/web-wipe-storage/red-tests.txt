            {countdown}
          </Text>
        ) : null}
      </View>
      <View style={styles.actions}>

        {lock.passcodeSet ? (
          <Button testID=\"unlock-passcode\" title={t(\"lock.usePasscode\")} variant={biometric ? \"link\" : \"cta\"} onPress={() => setPasscodeOpen(true)} />

}
